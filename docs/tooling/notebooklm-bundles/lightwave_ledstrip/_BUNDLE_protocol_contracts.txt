version: "1.0"
transport: "WebSocket JSON text frames"
endpoint: "ws://<k1-ip>:80/ws"
k1_ap_ip: "192.168.4.1"

# ===========================================================================
# Envelope Format
# ===========================================================================
envelope:
  request:
    type: "string (command name, REQUIRED; legacy clients may send cmd instead)"
    requestId: "string (optional correlation ID for request/response matching)"
  response:
    type: "string (echoes command name or response-specific type)"
    success: "bool (present on structured responses)"
    requestId: "string (echoed from request if provided)"
    data: "object (command-specific payload)"
  broadcast:
    type: "string (broadcast type name)"
    # Broadcasts have NO requestId. They are unsolicited K1 -> all clients.
  error:
    type: "string (echoes command name)"
    success: false
    error:
      code: "string (error code)"
      message: "string (human-readable)"

# queries. The host for every route in this section is the tab5 device's IP,

# Wave 2 hardening (2026-04-18): X-OTA-SHA256 is now REQUIRED on all tab5 OTA
# upload endpoints. Uploads without the header are rejected before Update.end()
# is called. The digest is computed over the received bytes and compared to the
# supplied value to guarantee image integrity.
#
# Tab5 OTA is a direct HTTP upload (multipart/form-data or
# application/octet-stream) — it is NOT framed over WebSocket.
# ---------------------------------------------------------------------------

    method: GET
    auth: false
    host: "tab5 (not K1)"
    description: "Get Tab5 firmware version and build info"

  /api/v1/firmware/update:
    method: POST
    host: "tab5 (not K1)"
    description: "Upload Tab5 firmware binary (OTA). As of Wave 2 (2026-04-18)
      X-OTA-SHA256 is REQUIRED — uploads without the header are rejected.
      SHA-256 is computed over received bytes and compared to the supplied
      digest before Update.end() is called."
    content_types: [ "multipart/form-data", "application/octet-stream" ]
